# ==============================
# 🧱 www BUILD STAGE
# ==============================
FROM node:20-bullseye-slim AS www_builder

WORKDIR /app

# Bersihkan cache agar build stabil
RUN rm -rf /app/* /root/.npm /root/.cache && npm cache clean --force

# Copy dependency manifest
COPY package*.json ./

# Install dependencies
RUN if [ -f package-lock.json ]; then \
    echo "🔒 Using npm ci"; \
    npm ci || (echo "⚠️ npm ci gagal, fallback ke npm install" && npm install); \
    else \
    echo "⚠️ package-lock.json tidak ditemukan, npm install"; \
    npm install; \
    fi

# Copy source code
COPY . .

# 🔥 FIX OOM: Tambah heap memory Node.js saat build
RUN NODE_OPTIONS="--max-old-space-size=4096" npm run build


# ==============================
# 🚀 www RUNTIME STAGE
# ==============================
FROM node:20-bullseye-slim AS www

WORKDIR /app

ENV NODE_ENV=development
ENV PORT=3000

# Copy hasil build saja (lebih kecil & aman)
COPY --from=www_builder /app/.next ./.next
COPY --from=www_builder /app/public ./public
COPY --from=www_builder /app/package*.json ./

# Install production dependencies saja
RUN rm -rf /root/.npm /root/.cache && npm cache clean --force && \
    if [ -f package-lock.json ]; then \
    echo "🔒 Installing prod deps via npm ci"; \
    npm ci --omit=dev || (echo "⚠️ npm ci gagal, fallback ke npm install" && npm install --omit=dev); \
    else \
    echo "⚠️ package-lock.json tidak ditemukan, pakai npm install"; \
    npm install --omit=dev; \
    fi && \
    npm cache clean --force

EXPOSE 3000

CMD ["npm", "start"]
